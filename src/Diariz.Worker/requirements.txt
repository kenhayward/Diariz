# torch / torchaudio are installed from the CUDA wheel index in the Dockerfile.
whisperx==3.8.6

# whisperx 3.8 targets pyannote.audio 4, and that is the point of this pairing rather than an incidental
# bump: the identical diarization pipeline on identical audio runs about 15x faster under pyannote 4
# than under 3.3.2 (measured on one 60 s window: 35.6s -> 2.3s). Diarization is ~79% of what a live
# chunk costs, so this is the difference between live transcription keeping up and pausing itself.
#
# It also dissolves the constraint the old pins existed for. whisperx 3.3.1 + pyannote 3.3.2 passed
# `use_auth_token` to hf_hub_download, which huggingface_hub 1.0 removed - so transformers and the hub
# had to be held below their majors. pyannote 4 does not pass it, so those ceilings are no longer load
# bearing; they are kept here only because whisperx 3.8.6 itself asks for `huggingface-hub<1.0.0`.
pyannote.audio>=4.0.0

# Still the pre-1.0 hub line, now because whisperx requires it rather than because pyannote does.
# transformers>=4.48 is whisperx's floor; 4.53.x remains the lowest 4.x clearing every fixable
# advisory (the ReDoS/input-validation set, all first patched at <=4.53.0). The three that remain need
# transformers>=5 and are dismissed as unreachable - all "attacker-controlled model repo / malicious
# checkpoint" paths, and this worker only ever loads fixed, server-configured models and never trains.
transformers==4.53.3
huggingface_hub==0.35.3

# Speaker embeddings (voiceprints): ECAPA-TDNN, 192-d, Apache-2.0 code + weights.
speechbrain==1.1.1

# Cap at the 3.10 line: the worker base image (nvidia/cuda ...-ubuntu22.04) ships Python 3.10, and
# matplotlib 3.11+ requires Python >= 3.11 (so 3.11.0 has no installable wheel here). Keep <3.11 until
# the base image's Python is bumped — a Dependabot bump to 3.11.x will break the Docker build.
matplotlib==3.10.9
redis==8.1.0
boto3==1.43.83
requests==2.34.2

# Optional error + performance reporting (GlitchTip / Sentry-compatible). Imported lazily and only
# when SENTRY_DSN is set, so it costs nothing on deployments that do not use it.
sentry-sdk==2.68.1
